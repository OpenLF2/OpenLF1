cmake_minimum_required(VERSION 3.20)
project(OpenLF1 VERSION 1.0.1 LANGUAGES CXX)
set(OPENLF1_VERSION_PATCH "${PROJECT_VERSION_PATCH}")
if(OPENLF1_VERSION_PATCH STREQUAL "")
    set(OPENLF1_VERSION_PATCH 0)
endif()
set(OPENLF1_COPYRIGHT_YEAR 2026)
if(MSVC)
    set(OPENLF1_WERROR /WX)
else()
    set(OPENLF1_WERROR -Werror)
endif()
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/OpenLF1Icons.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/dependencies.cmake)
option(OPENLF1_BUNDLE_DEPENDENCIES "Build dependencies as static libraries" OFF)
if(OPENLF1_BUNDLE_DEPENDENCIES)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    if(DJGPP)
        FetchContent_Declare(SDL3 URL ${OPENLF1_SDL3_DJGPP_URL})
    else()
        FetchContent_Declare(SDL3 URL ${OPENLF1_SDL3_RELEASE_URL})
    endif()
    FetchContent_Declare(zlib URL https://github.com/madler/zlib/releases/download/v1.3.1/zlib-1.3.1.tar.gz)
    set(SKIP_INSTALL_ALL ON)
    FetchContent_MakeAvailable(SDL3 zlib)
    unset(SKIP_INSTALL_ALL)
    set(OPENLF1_SDL SDL3::SDL3-static)
    set(OPENLF1_ZLIB zlibstatic)
else()
    find_package(SDL3 REQUIRED CONFIG)
    find_package(ZLIB REQUIRED)
    set(OPENLF1_SDL SDL3::SDL3)
    set(OPENLF1_ZLIB ZLIB::ZLIB)
endif()
set(OPENLF1_GAME_SOURCES
    src/combat.cpp src/field.cpp src/opponent.cpp src/roster.cpp src/round.cpp
    src/projectile.cpp src/tournament.cpp src/ground_item.cpp
    src/palette_cycle.cpp src/blast.cpp src/grapple.cpp)
add_executable(openlf1 src/main.cpp src/archive.cpp src/display.cpp ${OPENLF1_GAME_SOURCES})
target_compile_features(openlf1 PRIVATE cxx_std_23)
if(APPLE)
    set_target_properties(openlf1 PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_BUNDLE_NAME "OpenLF1"
        MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.openlf2.OpenLF1"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        MACOSX_BUNDLE_COPYRIGHT "Copyright (c) ${OPENLF1_COPYRIGHT_YEAR} OpenLF1")
    if(openlf1_icons)
        set_target_properties(openlf1 PROPERTIES MACOSX_BUNDLE_ICON_FILE "openlf1.icns")
        set(OPENLF1_MACOS_ICON "${openlf1_icons}/openlf1.icns")
        set_source_files_properties("${OPENLF1_MACOS_ICON}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
        target_sources(openlf1 PRIVATE "${OPENLF1_MACOS_ICON}")
    endif()
endif()
target_compile_options(openlf1 PRIVATE ${OPENLF1_WERROR})
if(openlf1_icons AND NOT DJGPP AND NOT EMSCRIPTEN)
    target_include_directories(openlf1 PRIVATE "${openlf1_icons}")
    target_compile_definitions(openlf1 PRIVATE OPENLF1_HAS_WINDOW_ICON=1)
endif()
if(WIN32)
    if(openlf1_icons)
        set(OPENLF1_ICON_RESOURCE "1 ICON \"${openlf1_icons}/openlf1.ico\"")
    else()
        set(OPENLF1_ICON_RESOURCE "")
    endif()
    configure_file("${PROJECT_SOURCE_DIR}/packaging/openlf1.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/openlf1.rc" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/openlf1.rc" PROPERTIES GENERATED TRUE)
    if(openlf1_icons)
        set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/openlf1.rc" APPEND PROPERTY
            OBJECT_DEPENDS "${openlf1_icons}/openlf1.ico")
    endif()
    enable_language(RC)
    target_sources(openlf1 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/openlf1.rc")
endif()
target_link_libraries(openlf1 PRIVATE ${OPENLF1_SDL} ${OPENLF1_ZLIB})
if(EMSCRIPTEN)
    target_compile_options(openlf1 PRIVATE -fexceptions)
    target_link_options(openlf1 PRIVATE
        -fexceptions
        -sFORCE_FILESYSTEM=1
        -sALLOW_MEMORY_GROWTH=1
        -sSTACK_SIZE=2097152
        -sASYNCIFY=1
        -sEXPORTED_RUNTIME_METHODS=FS,callMain)
endif()

if(UNIX AND NOT APPLE AND NOT EMSCRIPTEN AND NOT DJGPP)
    include(GNUInstallDirs)
    install(TARGETS openlf1 RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
    install(FILES packaging/openlf1.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
    install(FILES packaging/io.github.openlf2.OpenLF1.metainfo.xml DESTINATION "${CMAKE_INSTALL_DATADIR}/metainfo")
    install(FILES packaging/openlf1.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
    install(FILES LICENSE THIRD-PARTY-NOTICES.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/openlf1")
    set(CPACK_PACKAGE_NAME "openlf1")
    set(CPACK_PACKAGE_VENDOR "OpenLF1 contributors")
    set(CPACK_PACKAGE_CONTACT "OpenLF1 contributors <openlf1@users.noreply.github.com>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Open-source reimplementation of Little Fighter (1995)")
    set(CPACK_PACKAGE_DESCRIPTION "OpenLF1 is an open-source reimplementation of Little Fighter (1995). It uses the original game data to recreate the original's introduction, menus, character selection, and side-view battles.")
    set(CPACK_PACKAGE_HOMEPAGE_URL "https://openlf2.github.io/OpenLF1/")
    set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
    set(CPACK_DEBIAN_PACKAGE_NAME "openlf1")
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "${CPACK_PACKAGE_CONTACT}")
    set(CPACK_DEBIAN_PACKAGE_SECTION "games")
    set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
    set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "${CPACK_PACKAGE_HOMEPAGE_URL}")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}")
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_STRIP_FILES ON)
    include(CPack)
endif()
